    public int maxWalls(int[] robots, int[] distance, int[] walls) {
        if (robots.length == 1) {
            int a = 0;
            int b = 0;
            for (int wall : walls) {
                if (wall < robots[0] - distance[0] || wall > robots[0] + distance[0])
                    continue;
                if (wall < robots[0]) {
                    a++;
                } else if (wall > robots[0]) {
                    b++;
                } else {
                    a++;
                    b++;
                }
            }
            return Math.max(a, b);
        }

        }

        Arrays.sort(walls);
